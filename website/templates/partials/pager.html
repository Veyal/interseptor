{{- with .Page.Prev}} {{- $path := .Path}}{{$color := ""}}{{range $nav}}{{$group := .}}{{range .Links}}{{if eq .Path $path}}{{$color = $group.Color}}{{end}}{{end}}{{end}} {{icon "arrow-left"}}Previous {{with .Icon}}{{icon .}}{{end}}{{.Title}} {{- end}} {{- with .Page.Next}} {{- $path := .Path}}{{$color := ""}}{{range $nav}}{{$group := .}}{{range .Links}}{{if eq .Path $path}}{{$color = $group.Color}}{{end}}{{end}}{{end}} Next{{icon "arrow-right"}} {{with .Icon}}{{icon .}}{{end}}{{.Title}} {{- end}}
{{- end}} {{- end}}
